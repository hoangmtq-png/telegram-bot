def get_stock_count(cat_id):
    file_path = f"data/{cat_id}.txt"
    if not os.path.exists(file_path):
        return 0
    try:
        with open(file_path, "r", encoding="utf-8") as f:
            lines = [line.strip() for line in f.readlines() if line.strip()]
            return len(lines)
    except Exception:
        return 0

async def input_link_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    text = update.message.text.strip()
    selected_item = context.user_data.get("pending_service")

    if not selected_item:
        await update.message.reply_text("⚠️ Đã hết phiên. Bấm /start để làm lại.")
        return ConversationHandler.END

    is_cmt = selected_item.get("type") == "cmt"
    if is_cmt:
        context.user_data["order_cmt_content"] = text
    else:
        context.user_data["order_link"] = text

    await update.message.reply_text(
        f"🔢 **Nhập số lượng:**\n\n👇 Vui lòng nhập số lượng cần tăng (Ví dụ: 100, 500, 1000):",
        reply_markup=InlineKeyboardMarkup([[InlineKeyboardButton("❌ Hủy bỏ", callback_data="menu_main")]])
    )
    return INPUT_QUANTITY

async def input_quantity_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    text = update.message.text.strip()
    try:
        quantity = int(text)
        if quantity <= 0:
            raise ValueError()
    except ValueError:
        await update.message.reply_text("⚠️ Số lượng không hợp lệ! Vui lòng nhập số nguyên dương:")
        return INPUT_QUANTITY

    context.user_data["order_quantity"] = quantity

    # Bỏ qua phần chọn cảm xúc, chuyển thẳng sang bước nhập ghi chú
    await update.message.reply_text(
        "📝 **Ghi chú (Không bắt buộc):**\n\n👇 Vui lòng nhập ghi chú cho đơn hàng hoặc bấm nút [Bỏ qua ghi chú] bên dưới:",
        reply_markup=InlineKeyboardMarkup([[InlineKeyboardButton("⏩ Bỏ qua ghi chú", callback_data="skip_note")]])
    )
    return INPUT_NOTE

    await update.message.reply_text(confirm_text, reply_markup=InlineKeyboardMarkup(keyboard))

async def execute_buy_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    user = query.from_user
    data = query.data

    _, qty_str, price_str = data.split(":")
    quantity = int(qty_str)
    total_price = float(price_str)

    selected_item = context.user_data.get("pending_service")
    link = context.user_data.get("order_link", "N/A")
    user_balance = get_user_balance(user.id)

    if user_balance < total_price:
        await query.edit_message_text(
            "❌ **Số dư không đủ để thanh toán đơn hàng!**\n\n👇 Vui lòng nạp thêm tiền để tiếp tục:",
            reply_markup=InlineKeyboardMarkup([[InlineKeyboardButton("💳 Nạp Tiền Ngay", callback_data="nap_tien"), InlineKeyboardButton("🏠 Menu", callback_data="menu_main")]])
        )
        return

    USERS_DB[user.id]["balance"] -= total_price
    remaining = USERS_DB[user.id]["balance"]

    await query.edit_message_text(
        f"✅ **ĐẶT HÀNG THÀNH CÔNG!**\n"
        f"----------------------------------------\n"
        f"📌 Dịch vụ: {selected_item['name']}\n"
        f"🔢 Số lượng: {quantity:,}\n"
        f"💰 Số dư còn lại: {remaining:,.2f} VNĐ\n"
        f"----------------------------------------\n"
        f"⏳ Đơn hàng đã được chuyển tới Admin!",
        reply_markup=InlineKeyboardMarkup([[InlineKeyboardButton("🏠 Menu Chính", callback_data="menu_main")]])
    )

    admin_msg = (
        f"🔔 **CÓ ĐƠN HÀNG SMM MỚI!**\n"
        f"----------------------------------------\n"
        f"👤 Khách: {user.id} (@{user.username or 'No'})\n"
        f"📌 Dịch vụ: {selected_item['name']}\n"
        f"🔗 Link: {link}\n"
        f"🔢 Số lượng: {quantity:,}\n"
    )
    try:
        await context.bot.send_message(chat_id=ADMIN_CHAT_ID, text=admin_msg)
    except Exception:
        pass

async def input_topup_amount_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user = update.effective_user
    text = update.message.text.strip()
    try:
        amount = float(text)
        if amount <= 0:
            raise ValueError()
    except ValueError:
        await update.message.reply_text("⚠️ Số tiền không hợp lệ! Vui lòng nhập lại:")
        return INPUT_TOPUP_AMOUNT

    qr_url = f"https://img.vietqr.io/image/{BANK_INFO['bank_name']}-{BANK_INFO['account_no']}-compact2.png?amount={int(amount)}&addInfo={user.id}&accountName={BANK_INFO['account_name'].replace(' ', '%20')}"
    text_msg = f"🏦 Chuyển khoản {amount:,.0f}đ với nội dung: `{user.id}`"
    keyboard = [[InlineKeyboardButton("✅ Đã chuyển khoản", callback_data=f"confirm_trans:{int(amount)}")]]
    await context.bot.send_photo(chat_id=user.id, photo=qr_url, caption=text_msg, reply_markup=InlineKeyboardMarkup(keyboard))
    return ConversationHandler.END

def main():
    logging.basicConfig(format="%(asctime)s - %(name)s - %(levelname)s - %(message)s", level=logging.INFO)
    application = Application.builder().token(BOT_TOKEN).build()

    conv_handler = ConversationHandler(
        entry_points=[
            CallbackQueryHandler(button_handler, pattern="^custom_amount$|^buy:"),
        ],
        states={
            INPUT_LINK: [MessageHandler(filters.TEXT & ~filters.COMMAND, input_link_handler)],
            INPUT_QUANTITY: [MessageHandler(filters.TEXT & ~filters.COMMAND, input_quantity_handler)],
            INPUT_NOTE: [MessageHandler(filters.TEXT & ~filters.COMMAND, input_note_handler)],
            INPUT_TOPUP_AMOUNT: [MessageHandler(filters.TEXT & ~filters.COMMAND, input_topup_amount_handler)],
        },
        fallbacks=[CommandHandler("start", start), CallbackQueryHandler(button_handler, pattern="^menu_main$")],
    )

    application.add_handler(CommandHandler("start", start))
    application.add_handler(CommandHandler("topup", admin_topup_cmd))
    application.add_handler(conv_handler)
    application.add_handler(CallbackQueryHandler(skip_note_handler, pattern="^skip_note$"))
    application.add_handler(CallbackQueryHandler(execute_buy_handler, pattern="^execute_buy:"))

    application.run_polling()
